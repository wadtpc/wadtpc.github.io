/* ========== GALLERY LIGHTBOX WITH MULTI-PHOTOS ========== */ const lightbox = document.getElementById('lightbox'); const lightboxContent = document.getElementById('lightboxContent'); const lightboxCaption = document.getElementById('lightboxCaption'); const lightboxCounter = document.getElementById('lightboxCounter'); const lightboxClose = document.getElementById('lightboxClose'); const lightboxPrev = document.getElementById('lightboxPrev'); const lightboxNext = document.getElementById('lightboxNext'); // Gallery data — map folder name to array of image paths + captions const galleries = { church: { images: [ 'images/church/1.jpg', 'images/church/2.jpg', 'images/church/3.jpg', 'images/church/4.jpg' ], captions: [ 'With my church org', 'Community event', 'Serving together', 'Another moment' ] }, design: { images: [ 'images/design/1.jpg', 'images/design/2.jpg' ], captions: [ 'Figma mockup', 'Design experiment' ] }, convention: { images: [ 'images/convention/1.jpg', 'images/convention/2.jpg' ], captions: [ 'Convention planning', 'Team session' ] }, outreach: { images: [ 'images/outreach/1.jpg' ], captions: [ 'Outreach activity' ] }, workspace: { images: [ 'images/workspace/1.jpg' ], captions: [ 'My desk setup' ] }, presentation: { images: [ 'images/presentation/1.jpg' ], captions: [ 'Presenting MindCheck' ] } }; let currentGallery = []; let currentCaptions = []; let currentIndex = 0; let currentMainCaption = ''; function openGallery(galleryKey, mainCaption) { const gallery = galleries[galleryKey]; if (!gallery || gallery.images.length === 0) return; currentGallery = gallery.images; currentCaptions = gallery.captions; currentMainCaption = mainCaption || ''; currentIndex = 0; renderLightboxImage(); lightbox.classList.add('active'); document.body.style.overflow = 'hidden'; } function renderLightboxImage() { const src = currentGallery[currentIndex]; const caption = currentCaptions[currentIndex] || currentMainCaption; lightboxContent.innerHTML = ``; lightboxCaption.textContent = caption; lightboxCounter.textContent = `${currentIndex + 1} of ${currentGallery.length}`; // Show/hide nav buttons based on count const showNav = currentGallery.length > 1; lightboxPrev.style.display = showNav ? 'flex' : 'none'; lightboxNext.style.display = showNav ? 'flex' : 'none'; } function nextImage() { if (currentGallery.length <= 1) return; currentIndex = (currentIndex + 1) % currentGallery.length; renderLightboxImage(); } function prevImage() { if (currentGallery.length <= 1) return; currentIndex = (currentIndex - 1 + currentGallery.length) % currentGallery.length; renderLightboxImage(); } function closeLightbox() { lightbox.classList.remove('active'); document.body.style.overflow = ''; } document.querySelectorAll('.gallery-tile').forEach(tile => { tile.addEventListener('click', () => { const galleryKey = tile.dataset.gallery; const mainCaption = tile.dataset.caption || ''; openGallery(galleryKey, mainCaption); }); }); lightboxClose.addEventListener('click', closeLightbox); lightboxPrev.addEventListener('click', (e) => { e.stopPropagation(); prevImage(); }); lightboxNext.addEventListener('click', (e) => { e.stopPropagation(); nextImage(); }); lightbox.addEventListener('click', (e) => { if (e.target === lightbox) closeLightbox(); }); document.addEventListener('keydown', (e) => { if (!lightbox.classList.contains('active')) return; if (e.key === 'Escape') closeLightbox(); if (e.key === 'ArrowRight') nextImage(); if (e.key === 'ArrowLeft') prevImage(); });
